   127→      });
   128→      
   129→      const data = await res.json();
   130→      
   131→      if (!res.ok) throw new Error(data.error || `HTTP ${res.status}`);
   132→      if (data.success) {
   133→        setIpResult(data.data);
   134→        setSuccess(`IP ${data.data.query} analyzed - Risk Level: ${data.data.threat.level}`);
   135→      } else throw new Error(data.error);

   137→      setError(err.message);
   138→    }
