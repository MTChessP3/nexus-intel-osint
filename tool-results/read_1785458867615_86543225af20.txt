     1→'use client';
     2→
     3→import React, { useState, useEffect, useCallback } from 'react';
     4→import { Card, CardContent, CardHeader, CardTitle } from '@/components/ui/card';
     5→import { Button } from '@/components/ui/button';
     6→import { Input } from '@/components/ui/input';
     7→import { Badge } from '@/components/ui/badge';
     8→import { Tabs, TabsContent, TabsList, TabsTrigger } from '@/components/ui/tabs';
     9→import { Separator } from '@/components/ui/separator';
    10→import { ScrollArea } from '@/components/ui/scroll-area';
    11→import { 
    12→  Shield, Search, Globe, AlertTriangle, CheckCircle2, 
    13→  XCircle, Activity, FileText, Download, ExternalLink,
    14→  MapPin, Server, Fingerprint, Bug, Link, Hash, Target,
    15→  Clock, Database, Network, Terminal, BarChart3,
    16→  RefreshCw, ChevronRight, Info
    17→} from 'lucide-react';
    18→
    19→// ============= TYPES =============
    20→interface IPResult {
    21→  query: string;
    22→  geolocation: {
    23→    country: string; countryCode: string; region: string;
    24→    city: string; latitude: number; longitude: number; timezone: string;
    25→  };
    26→  network: {
    27→    isp: string; org: string; asn: string; asFull: string;
    28→    isMobile: boolean; isProxy: boolean; isHosting: boolean;
    29→  };
    30→  threat: {
    31→    score: number; level: string; color: string;
    32→    indicators: string[]; recommendations: string[];
    33→  };
    34→}
    35→
    36→interface DomainResult {
    37→  domain: string;
    38→  whois: any;
    39→  dns: any;
    40→  security: any;
    41→  reputation: {
    42→    score: number; level: string; color: string;
    43→    indicators: string[]; recommendations: string[];
    44→  };
    45→}
    46→
    47→interface CVEResult {
    48→  id: string;
    49→  descriptions: string;
    50→  cvss: { score: number | null; severity: string; vector: string; version: string };
    51→  cwe: string[];
    52→  references: any[];
    53→  dates: any;
    54→  status: string;
    55→}
    56→
    57→interface URLResult {
    58→  url: string;
    59→  parsedUrl: any;
    60→  overallAssessment: {
    61→    threatScore: number; riskLevel: string; riskColor: string;
    62→    indicators: string[]; recommendations: string[]; verdict: string;
    63→  };
    64→}
    65→
    66→interface HashResult {
    67→  input: { hash: string; hashType: string };
    68→  found: boolean;
    69→  aggregateResults: {
    70→    detectionRate: number; classification: string;
    71→    threatLevel: string; color: string;
    72→  };
    73→  engineResults: any[];
    74→  details?: any;
    75→}
    76→
    77→interface ThreatData {
    78→  iocs: any[];
    79→  activeThreats: any[];
    80→  campaigns: any[];
    81→  aptGroups: any[];
    82→  globalThreatLevel: any;
    83→  statistics: any;
    84→}
    85→
    86→// ============= MAIN COMPONENT =============
    87→export default function NexusIntelDashboard() {
    88→  const [activeTab, setActiveTab] = useState('dashboard');
    89→  const [isLoading, setIsLoading] = useState(false);
    90→  const [error, setError] = useState<string | null>(null);
    91→  
    92→  // States
    93→  const [ipInput, setIpInput] = useState('');
    94→  const [ipResult, setIpResult] = useState<IPResult | null>(null);
    95→  const [domainInput, setDomainInput] = useState('');
    96→  const [domainResult, setDomainResult] = useState<DomainResult | null>(null);
    97→  const [cveInput, setCveInput] = useState('');
    98→  const [cveResults, setCveResults] = useState<CVEResult[]>([]);
    99→  const [urlInput, setUrlInput] = useState('');
   100→  const [urlResult, setUrlResult] = useState<URLResult | null>(null);
   101→  const [hashInput, setHashInput] = useState('');
   102→  const [hashResult, setHashResult] = useState<HashResult | null>(null);
   103→  const [threatData, setThreatData] = useState<ThreatData | null>(null);
   104→  const [reportData, setReportData] = useState<any>(null);
   105→
   106→  // API Functions
   107→  const fetchThreatIntelligence = useCallback(async () => {
   108→    try {
   109→      setError(null);
   110→      const res = await fetch('/api/osint/threats');
   111→      const data = await res.json();
   112→      if (data.success) setThreatData(data.data);
   113→      else throw new Error(data.error || 'Error fetching threats');
   114→    } catch (err: any) {
   115→      console.error('Threat intel error:', err);
   116→      setError(`Error: ${err.message}`);
   117→    }
   118→  }, []);
   119→
   120→  useEffect(() => {
   121→    fetchThreatIntelligence();
   122→  }, [fetchThreatIntelligence]);
   123→
   124→  const analyzeIP = async () => {
   125→    if (!ipInput) return;
   126→    setIsLoading(true); setError(null);
   127→    try {
   128→      const res = await fetch('/api/osint/ip', {
   129→        method: 'POST',
   130→        headers: { 'Content-Type': 'application/json' },
   131→        body: JSON.stringify({ ip: ipInput.trim() })
   132→      });
   133→      const data = await res.json();
   134→      if (data.success) setIpResult(data.data);
   135→      else throw new Error(data.error || 'Error analyzing IP');
   136→    } catch (err: any) {
   137→      setError(err.message); setIpResult(null);
   138→    }
   139→    setIsLoading(false);
   140→  };
   141→
   142→  const analyzeDomain = async () => {
   143→    if (!domainInput) return;
   144→    setIsLoading(true); setError(null);
   145→    try {
   146→      const res = await fetch('/api/osint/domain', {
   147→        method: 'POST', headers: { 'Content-Type': 'application/json' },
   148→        body: JSON.stringify({ domain: domainInput.trim() })
   149→      });
   150→      const data = await res.json();
   151→      if (data.success) setDomainResult(data.data);
   152→      else throw new Error(data.error);
   153→    } catch (err: any) { setError(err.message); setDomainResult(null); }
   154→    setIsLoading(false);
   155→  };
   156→
   157→  const searchCVE = async () => {
   158→    if (!cveInput) return;
   159→    setIsLoading(true); setError(null);
   160→    try {
   161→      const isCVEId = cveInput.toUpperCase().startsWith('CVE-');
   162→      const res = await fetch('/api/osint/cve', {
   163→        method: 'POST', headers: { 'Content-Type': 'application/json' },
   164→        body: JSON.stringify(isCVEId ? { cveId: cveInput } : { keyword: cveInput })
   165→      });
   166→      const data = await res.json();
   167→      if (data.success) setCveResults(data.data.results || [data.data]);
   168→      else throw new Error(data.error);
   169→    } catch (err: any) { setError(err.message); setCveResults([]); }
   170→    setIsLoading(false);
   171→  };
   172→
   173→  const analyzeURL = async () => {
   174→    if (!urlInput) return;
   175→    setIsLoading(true); setError(null);
   176→    try {
   177→      const res = await fetch('/api/osint/url', {
   178→        method: 'POST', headers: { 'Content-Type': 'application/json' },
   179→        body: JSON.stringify({ url: urlInput.trim() })
   180→      });
   181→      const data = await res.json();
   182→      if (data.success) setUrlResult(data.data);
   183→      else throw new Error(data.error);
   184→    } catch (err: any) { setError(err.message); setUrlResult(null); }
   185→    setIsLoading(false);
   186→  };
   187→
   188→  const analyzeHash = async () => {
   189→    if (!hashInput) return;
   190→    setIsLoading(true); setError(null);
   191→    try {
   192→      const res = await fetch('/api/osint/hash', {
   193→        method: 'POST', headers: { 'Content-Type': 'application/json' },
   194→        body: JSON.stringify({ hash: hashInput.trim() })
   195→      });
   196→      const data = await res.json();
   197→      if (data.success) setHashResult(data.data);
   198→      else throw new Error(data.error);
   199→    } catch (err: any) { setError(err.message); setHashResult(null); }
   200→    setIsLoading(false);
   201→  };
   202→
   203→  const generateReport = async (type: string) => {
   204→    setIsLoading(true); setError(null);
   205→    try {
   206→      const res = await fetch('/api/osint/reports', {
   207→        method: 'POST', headers: { 'Content-Type': 'application/json' },
   208→        body: JSON.stringify({ reportType: type, data: threatData || {}, options: { classification: 'CONFIDENTIAL' } })
   209→      });
   210→      const data = await res.json();
   211→      if (data.success) setReportData(data.data.report);
   212→      else throw new Error(data.error);
   213→    } catch (err: any) { setError(err.message); }
   214→    setIsLoading(false);
   215→  };
   216→
   217→  // Professional severity colors (muted, enterprise-style)
   218→  const getSeverityStyle = (level: string) => {
   219→    switch (level?.toUpperCase()) {
   220→      case 'CRITICAL':
   221→      case 'CRÍTICO':
   222→      case 'PELIGROSO':
   223→      case 'MALICIOUS':
   224→        return 'bg-red-100 text-red-800 border-red-200';
   225→      case 'HIGH':
   226→      case 'ALTO':
   227→      case 'ALTO RIESGO':
   228→      case 'SUSPICIOUS':
   229→        return 'bg-orange-100 text-orange-800 border-orange-200';
   230→      case 'MEDIUM':
   231→      case 'MEDIO':
   232→      case 'SOSPECHOSO':
   233→      case 'CAUTELA':
   234→        return 'bg-yellow-100 text-yellow-800 border-yellow-200';
   235→      default:
   236→        return 'bg-green-100 text-green-800 border-green-200';
   237→    }
   238→  };
   239→
   240→  return (
   241→    <div className="min-h-screen bg-gray-50">
   242→      {/* Professional Header */}
   243→      <header className="bg-white border-b border-gray-200 sticky top-0 z-50">
   244→        <div className="max-w-7xl mx-auto px-4 sm:px-6 lg:px-8">
   245→          <div className="flex items-center justify-between h-16">
   246→            <div className="flex items-center gap-3">
   247→              <div className="w-9 h-9 bg-gray-900 rounded-lg flex items-center justify-center">
   248→                <Shield className="h-5 w-5 text-white" />
   249→              </div>
   250→              <div>
   251→                <h1 className="text-lg font-semibold text-gray-900">NEXUS INTEL</h1>
   252→                <p className="text-xs text-gray-500">OSINT Threat Intelligence Platform</p>
   253→              </div>
   254→            </div>
   255→            
   256→            <div className="flex items-center gap-4">
   257→              {threatData?.globalThreatLevel && (
   258→                <span className={`inline-flex items-center gap-1.5 px-3 py-1 rounded-full text-xs font-medium border ${getSeverityStyle(threatData.globalThreatLevel.level)}`}>
   259→                  <Activity className="h-3 w-3" />
   260→                  Threat Level: {threatData.globalThreatLevel.level} ({threatData.globalThreatLevel.score})
   261→                </span>
   262→              )}
   263→              <Button variant="ghost" size="sm" onClick={fetchThreatIntelligence} disabled={isLoading}>
   264→                <RefreshCw className={`h-4 w-4 ${isLoading ? 'animate-spin' : ''}`} />
   265→              </Button>
   266→            </div>
   267→          </div>
   268→        </div>
   269→      </header>
   270→
   271→      {/* Main Content */}
   272→      <main className="max-w-7xl mx-auto px-4 sm:px-6 lg:px-8 py-6">
   273→        {/* Error Banner */}
   274→        {error && (
   275→          <div className="mb-4 p-3 bg-red-50 border border-red-200 rounded-lg flex items-start gap-2">
   276→            <AlertTriangle className="h-5 w-5 text-red-600 mt-0.5 shrink-0" />
   277→            <div className="flex-1">
   278→              <p className="text-sm text-red-800">{error}</p>
   279→            </div>
   280→            <Button variant="ghost" size="sm" onClick={() => setError(null)} className="text-red-600 hover:text-red-800">
   281→              <XCircle className="h-4 w-4" />
   282→            </Button>
   283→          </div>
   284→        )}
   285→
   286→        <Tabs value={activeTab} onValueChange={setActiveTab} className="space-y-6">
   287→          <TabsList className="bg-white border border-gray-200 p-1 h-auto grid grid-cols-5 lg:grid-cols-10 gap-1">
   288→            {[
   289→              { value: 'dashboard', label: 'Dashboard', icon: Activity },
   290→              { value: 'ip', label: 'IP Intel', icon: Globe },
   291→              { value: 'domain', label: 'Domain', icon: Globe },
   292→              { value: 'cve', label: 'CVE', icon: Bug },
   293→              { value: 'url', label: 'URL', icon: Link },
   294→              { value: 'hash', label: 'Hash', icon: Fingerprint },
   295→              { value: 'threats', label: 'Threats', icon: Target },
   296→              { value: 'reports', label: 'Reports', icon: FileText },
   297→              { value: 'monitoring', label: 'Monitor', icon: Terminal },
   298→              { value: 'darkweb', label: 'DarkWeb', icon: Eye }
   299→            ].map(tab => (
   300→              <TabsTrigger 
   301→                key={tab.value} 
   302→                value={tab.value} 
   303→                className="text-xs py-2 data-[state=active]:bg-gray-900 data-[state=active]:text-white"
   304→              >
   305→                <tab.icon className="h-3 w-3 mr-1 hidden sm:inline" />
   306→                {tab.label}
   307→              </TabsTrigger>
   308→            ))}
   309→          </TabsList>
   310→
   311→          {/* ==================== DASHBOARD ==================== */}
   312→          <TabsContent value="dashboard" className="space-y-6">
   313→            {/* KPI Cards */}
   314→            <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-4 gap-4">
   315→              <Card>
   316→                <CardHeader className="pb-2">
   317→                  <p className="text-xs font-medium text-gray-500 uppercase tracking-wide">Global Threat Level</p>
   318→                </CardHeader>
   319→                <CardContent>
   320→                  {threatData?.globalThreatLevel ? (
   321→                    <div className="flex items-baseline gap-2">
   322→                      <span className="text-3xl font-bold text-gray-900">{threatData.globalThreatLevel.score}</span>
   323→                      <span className={`px-2 py-0.5 rounded text-xs font-medium ${getSeverityStyle(threatData.globalThreatLevel.level)}`}>
   324→                        {threatData.globalThreatLevel.level}
   325→                      </span>
   326→                    </div>
   327→                  ) : (
   328→                    <div className="animate-pulse h-10 bg-gray-200 rounded"></div>
   329→                  )}
   330→                  <p className="text-xs text-gray-500 mt-1">out of 100</p>
   331→                </CardContent>
   332→              </Card>
   333→
   334→              <Card>
   335→                <CardHeader className="pb-2"><p className="text-xs font-medium text-gray-500 uppercase">Active IOCs</p></CardHeader>
   336→                <CardContent>
   337→                  <span className="text-3xl font-bold text-gray-900">{threatData?.iocs?.length || 0}</span>
   338→                  <p className="text-xs text-gray-500 mt-1">indicators monitored</p>
   339→                </CardContent>
   340→              </Card>
   341→
   342→              <Card>
   343→                <CardHeader className="pb-2"><p className="text-xs font-medium text-gray-500 uppercase">Recent CVEs</p></CardHeader>
   344→                <CardContent>
   345→                  <span className="text-3xl font-bold text-gray-900">{threatData?.activeThreats?.length || 0}</span>
   346→                  <p className="text-xs text-gray-500 mt-1">from NVD database</p>
   347→                </CardContent>
   348→              </Card>
   349→
   350→              <Card>
   351→                <CardHeader className="pb-2"><p className="text-xs font-medium text-gray-500 uppercase">Active Campaigns</p></CardHeader>
   352→                <CardContent>
   353→                  <span className="text-3xl font-bold text-red-700">
   354→                    {threatData?.campaigns?.filter((c: any) => c.status === 'ACTIVE').length || 0}
   355→                  </span>
   356→                  <p className="text-xs text-gray-500 mt-1">require attention</p>
   357→                </CardContent>
   358→              </Card>
   359→            </div>
   360→
   361→            {/* Quick Analysis */}
   362→            <Card>
   363→              <CardHeader>
   364→                <CardTitle className="text-base">Quick Analysis</CardTitle>
   365→              </CardHeader>
   366→              <CardContent>
   367→                <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-4 gap-4">
   368→                  <div className="space-y-1.5">
   369→                    <label className="text-xs font-medium text-gray-600">IP Address</label>
   370→                    <div className="flex gap-2">
   371→                      <Input placeholder="8.8.8.8" value={ipInput} onChange={(e) => setIpInput(e.target.value)}
   372→                        onKeyDown={(e) => e.key === 'Enter' && analyzeIP()} className="text-sm" />
   373→                      <Button size="sm" onClick={analyzeIP} disabled={isLoading}><Search className="h-4 w-4" /></Button>
   374→                    </div>
   375→                  </div>
   376→                  <div className="space-y-1.5">
   377→                    <label className="text-xs font-medium text-gray-600">Domain</label>
   378→                    <div className="flex gap-2">
   379→                      <Input placeholder="example.com" value={domainInput} onChange={(e) => setDomainInput(e.target.value)}
   380→                        onKeyDown={(e) => e.key === 'Enter' && analyzeDomain()} className="text-sm" />
   381→                      <Button size="sm" onClick={analyzeDomain} disabled={isLoading}><Search className="h-4 w-4" /></Button>
   382→                    </div>
   383→                  </div>
   384→                  <div className="space-y-1.5">
   385→                    <label className="text-xs font-medium text-gray-600">CVE ID / Keyword</label>
   386→                    <div className="flex gap-2">
   387→                      <Input placeholder="CVE-2024-1234" value={cveInput} onChange={(e) => setCveInput(e.target.value)}
   388→                        onKeyDown={(e) => e.key === 'Enter' && searchCVE()} className="text-sm" />
   389→                      <Button size="sm" onClick={searchCVE} disabled={isLoading}><Search className="h-4 w-4" /></Button>
   390→                    </div>
   391→                  </div>
   392→                  <div className="space-y-1.5">
   393→                    <label className="text-xs font-medium text-gray-600">URL</label>
   394→                    <div className="flex gap-2">
   395→                      <Input placeholder="https://..." value={urlInput} onChange={(e) => setUrlInput(e.target.value)}
   396→                        onKeyDown={(e) => e.key === 'Enter' && analyzeURL()} className="text-sm" />
   397→                      <Button size="sm" onClick={analyzeURL} disabled={isLoading}><Search className="h-4 w-4" /></Button>
   398→                    </div>
   399→                  </div>
   400→                </div>
   401→
   402→                {/* Latest Result */}
   403→                {(ipResult || domainResult || urlResult || hashResult) && (
   404→                  <div className="mt-4 p-4 bg-gray-50 rounded-lg border border-gray-200">
   405→                    <h4 className="text-sm font-medium text-gray-700 mb-2">Latest Result</h4>
   406→                    
   407→                    {ipResult && (
   408→                      <div className="space-y-2">
   409→                        <div className="flex items-center gap-2 flex-wrap">
   410→                          <code className="text-sm bg-white px-2 py-0.5 rounded border">{ipResult.query}</code>
   411→                          <span className={`px-2 py-0.5 rounded text-xs font-medium ${getSeverityStyle(ipResult.threat.level)}`}>
   412→                            {ipResult.threat.level}: {ipResult.threat.score}/100
   413→                          </span>
   414→                        </div>
   415→                        <p className="text-xs text-gray-600">
   416→                          Location: {ipResult.geolocation.city}, {ipResult.geolocation.country} | ISP: {ipResult.network.isp}
   417→                        </p>
   418→                      </div>
   419→                    )}
   420→                    
   421→                    {urlResult && (
   422→                      <div className="space-y-2">
   423→                        <div className="flex items-center gap-2 flex-wrap">
   424→                          <code className="text-sm bg-white px-2 py-0.5 rounded border truncate max-w-md">{urlResult.url}</code>
   425→                          <span className={`px-2 py-0.5 rounded text-xs font-medium ${getSeverityStyle(urlResult.overallAssessment.riskLevel)}`}>
   426→                            {urlResult.overallAssessment.riskLevel}
   427→                          </span>
   428→                        </div>
   429→                        <p className="text-xs text-gray-600">{urlResult.overallAssessment.verdict}</p>
   430→                      </div>
   431→                    )}
   432→                  </div>
   433→                )}
   434→              </CardContent>
   435→            </Card>
   436→
   437→            {/* Recent Threats from NVD */}
   438→            {threatData?.activeThreats && threatData.activeThreats.length > 0 && (
   439→              <Card>
   440→                <CardHeader className="flex flex-row items-center justify-between">
   441→                  <CardTitle className="text-base">Recent Vulnerabilities (NIST NVD)</CardTitle>
   442→                  <Button variant="ghost" size="sm" onClick={() => setActiveTab('cve')}>
   443→                    View all <ChevronRight className="h-4 w-4 ml-1" />
   444→                  </Button>
   445→                </CardHeader>
   446→                <CardContent>
   447→                  <div className="overflow-x-auto">
   448→                    <table className="w-full text-sm">
   449→                      <thead>
   450→                        <tr className="border-b border-gray-200">
   451→                          <th className="text-left py-2 px-2 font-medium text-gray-600">CVE ID</th>
   452→                          <th className="text-left py-2 px-2 font-medium text-gray-600">Severity</th>
   453→                          <th className="text-left py-2 px-2 font-medium text-gray-600">CVSS</th>
   454→                          <th className="text-left py-2 px-2 font-medium text-gray-600">Published</th>
   455→                        </tr>
   456→                      </thead>
   457→                      <tbody>
   458→                        {threatData.activeThreats.slice(0, 5).map((t: any, i: number) => (
   459→                          <tr key={i} className="border-b border-gray-100 hover:bg-gray-50">
   460→                            <td className="py-2 px-2 font-mono text-blue-700">{t.id}</td>
   461→                            <td className="py-2 px-2">
   462→                              <span className={`px-2 py-0.5 rounded text-xs ${getSeverityStyle(t.severity)}`}>{t.severity}</span>
   463→                            </td>
   464→                            <td className="py-2 px-2">{t.cvssScore || '-'}</td>
   465→                            <td className="py-2 px-2 text-gray-500">{new Date(t.published).toLocaleDateString()}</td>
   466→                          </tr>
   467→                        ))}
   468→                      </tbody>
   469→                    </table>
   470→                  </div>
   471→                </CardContent>
   472→              </Card>
   473→            )}
   474→          </TabsContent>
   475→
   476→          {/* ==================== IP INTEL ==================== */}
   477→          <TabsContent value="ip" className="space-y-6">
   478→            <Card>
   479→              <CardHeader>
   480→                <CardTitle className="text-base flex items-center gap-2">
   481→                  <Globe className="h-5 w-5" /> IP Intelligence Analysis
   482→                </CardTitle>
   483→                <p className="text-sm text-gray-600">Geolocation and threat assessment using ip-api.com</p>
   484→              </CardHeader>
   485→              <CardContent className="space-y-4">
