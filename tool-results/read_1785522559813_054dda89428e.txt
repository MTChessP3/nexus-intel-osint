   503→  const formatTimestamp = (ts: string): string => {
   504→    if (!ts) return 'N/A';
   505→    const date = new Date(ts);
   506→    const now = new Date();
   507→    const diffMs = now.getTime() - date.getTime();
   508→    const diffMins = Math.floor(diffMs / 60000);
   509→    const diffHours = Math.floor(diffMs / 3600000);
   510→    const diffDays = Math.floor(diffMs / 86400000);
   511→    
   512→    if (diffMins < 1) return 'Just now';
   513→    if (diffMins < 60) return `${diffMins}m ago`;
   514→    if (diffHours < 24) return `${diffHours}h ago`;
   515→    if (diffDays < 7) return `${diffDays}d ago`;
   516→    return date.toLocaleDateString('en-US', { month: 'short', day: 'numeric' });
   517→  };
