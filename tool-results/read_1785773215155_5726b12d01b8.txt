    77→// Live Threat Timeline Data Generator
    78→const generateTimelineData = (): TimelineEvent[] => {
    79→  const events: TimelineEvent[] = [];
    80→  const now = new Date();
    81→  
    82→  const threatTypes = [
    83→    { event: 'New malware sample detected in wild', type: 'threat' as const, severity: 'HIGH' as Severity },
    84→    { event: 'Suspicious IP address flagged by sensors', type: 'ioc' as const, severity: 'MEDIUM' as Severity },
    85→    { event: 'Domain resolution change detected', type: 'analysis' as const, severity: 'LOW' as Severity },
    86→    { event: 'Threat feed update received (CISA)', type: 'system' as const },
    87→    { event: 'Critical CVE published to NVD', type: 'alert' as const, severity: 'CRITICAL' as Severity },
    88→    { event: 'Phishing campaign targeting finance sector', type: 'threat' as const, severity: 'HIGH' as Severity },
    89→    { event: 'IOC added to watchlist', type: 'ioc' as const, severity: 'MEDIUM' as Severity },
    90→    { event: 'Dark web marketplace activity spike', type: 'threat' as const, severity: 'HIGH' as Severity },
    91→    { event: 'SSL certificate expiration warning', type: 'alert' as const, severity: 'LOW' as Severity },
    92→    { event: 'Botnet C2 communication detected', type: 'threat' as const, severity: 'CRITICAL' as Severity },
    93→    { event: 'Data breach notification received', type: 'alert' as const, severity: 'CRITICAL' as Severity },
    94→    { event: 'Zero-day exploit being traded', type: 'threat' as const, severity: 'CRITICAL' as Severity }
    95→  ];
    96→  
    97→  for (let i = 0; i < 20; i++) {
    98→    const timeOffset = Math.floor(Math.random() * 60);
    99→    const threat = threatTypes[Math.floor(Math.random() * threatTypes.length)];
   100→    events.push({
   101→      time: new Date(now.getTime() - timeOffset * 60000).toLocaleTimeString(),
   102→      ...threat
   103→    });
   104→  }
   105→  
   106→  return events.sort((a, b) => new Date(`2000/01/01 ${b.time}`).getTime() - new Date(`2000/01/01 ${a.time}`).getTime());
   107→};
